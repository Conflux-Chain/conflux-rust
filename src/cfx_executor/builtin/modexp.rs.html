20// calculate modexp: left-to-right binary exponentiation to keep multiplicands
21// lower
22pub fn modexp(mut base: BigUint, exp: Vec<u8>, modulus: BigUint) -> BigUint {
23    const BITS_PER_DIGIT: usize = 8;
24
25    // n^m % 0 || n^m % 1
26    if modulus <= BigUint::one() {
27        return BigUint::zero();
28    }
29
30    // normalize exponent
31    let mut exp = exp.into_iter().skip_while(|d| *d == 0).peekable();
32
33    // n^0 % m
34    if let None = exp.peek() {
35        return BigUint::one();
36    }
37
38    // 0^n % m, n > 0
39    if base.is_zero() {
40        return BigUint::zero();
41    }
42
43    base = base % &modulus;
44
45    // Fast path for base divisible by modulus.
46    if base.is_zero() {
47        return BigUint::zero();
48    }
49
50    // Left-to-right binary exponentiation (Handbook of Applied Cryptography -
51    // Algorithm 14.79). http://www.cacr.math.uwaterloo.ca/hac/about/chap14.pdf
52    let mut result = BigUint::one();
53
54    for digit in exp {
55        let mut mask = 1 << (BITS_PER_DIGIT - 1);
56
57        for _ in 0..BITS_PER_DIGIT {
58            result = &result * &result % &modulus;
59
60            if digit & mask > 0 {
61                result = result * &base % &modulus;
62            }
63
64            mask >>= 1;
65        }
66    }
67
68    result
69}

71impl Precompile for ModexpImpl {
72    fn execute(
73        &self, input: &[u8], output: &mut BytesRef,
74    ) -> Result<(), Error> {
75        let mut reader = input.chain(io::repeat(0));
76        let mut buf = [0; 32];
77
78        // read lengths as usize.
79        // ignoring the first 24 bytes might technically lead us to fall out of
80        // consensus, but so would running out of addressable memory!
81        let mut read_len = |reader: &mut io::Chain<&[u8], io::Repeat>| {
82            reader
83                .read_exact(&mut buf[..])
84                .expect("reading from zero-extended memory cannot fail; qed");
85            BigEndian::read_u64(&buf[24..]) as usize
86        };
87
88        let base_len = read_len(&mut reader);
89        let exp_len = read_len(&mut reader);
90        let mod_len = read_len(&mut reader);
91
92        // Gas formula allows arbitrary large exp_len when base and modulus are
93        // empty, so we need to handle empty base first.
94        let r = if base_len == 0 && mod_len == 0 {
95            BigUint::zero()
96        } else {
97            // read the numbers themselves.
98            let mut buf = vec![0; max(mod_len, max(base_len, exp_len))];
99            let mut read_num = |reader: &mut io::Chain<&[u8], io::Repeat>,
100                                len: usize| {
101                reader.read_exact(&mut buf[..len]).expect(
102                    "reading from zero-extended memory cannot fail; qed",
103                );
104                BigUint::from_bytes_be(&buf[..len])
105            };
106
107            let base = read_num(&mut reader, base_len);
108
109            let mut exp_buf = vec![0; exp_len];
110            reader
111                .read_exact(&mut exp_buf[..exp_len])
112                .expect("reading from zero-extended memory cannot fail; qed");
113
114            let modulus = read_num(&mut reader, mod_len);
115
116            modexp(base, exp_buf, modulus)
117        };
118
119        // write output to given memory, left padded and same length as the
120        // modulus.
121        let bytes = r.to_bytes_be();
122
123        // always true except in the case of zero-length modulus, which leads to
124        // output of length and value 1.
125        if bytes.len() <= mod_len {
126            let res_start = mod_len - bytes.len();
127            output.write(res_start, &bytes);
128        }
129
130        Ok(())
131    }
132}

134/// A special pricing model for modular exponentiation.

136pub(crate) enum ModexpPricer {
137    Byzantium { divisor: usize },
138    // CIP-645e: EIP-2565
139    Berlin { base: usize },
140    // CIP-174: EIP-7823 + EIP-7883
141    Osaka { base: usize },
142}

146        ModexpPricer::Byzantium { divisor }

150        ModexpPricer::Berlin { base }
151    }

153    pub(crate) fn new_osaka(base: usize) -> ModexpPricer {

155    }

178impl PricePlan for ModexpPricePlan {
179    fn pricer(&self, spec: &Spec) -> &dyn Pricer {
180        if spec.cip174 {
181            &self.osaka
182        } else if spec.cip645.eip2565 {
183            &self.berlin
184        } else {
185            &self.byzantium
186        }

190impl Pricer for ModexpPricer {
191    fn cost(&self, input: &[u8]) -> U256 {
192        let mut reader = input.chain(io::repeat(0));
193        let mut buf = [0; 32];
194
195        // read lengths as U256 here for accurate gas calculation.
196        let mut read_len = || {
197            reader
198                .read_exact(&mut buf[..])
199                .expect("reading from zero-extended memory cannot fail; qed");
200            U256::from_big_endian(&buf[..])
201        };
202        let base_len = read_len();
203        let exp_len = read_len();
204        let mod_len = read_len();
205
206        // EIP-7823: over-limit inputs make the precompile fail; returning
207        // the maximal cost makes the call run out of gas, which consumes
208        // all gas passed to the call as the EIP requires.
209        if matches!(self, Self::Osaka { .. }) {
210            let limit = U256::from(EIP7823_INPUT_LEN_LIMIT);
211            if base_len > limit || exp_len > limit || mod_len > limit {
212                return U256::max_value();
213            }
214        }
215
216        if mod_len.is_zero() && base_len.is_zero() {
217            match self {
218                Self::Byzantium { .. } => return 0.into(),
219                Self::Berlin { base } => return (*base).into(),
220                // EIP-7883: multiplication complexity is at least 16, so a
221                // long exponent still charges iteration costs even with
222                // empty base and modulus — fall through to the full formula.
223                Self::Osaka { .. } => {}
224            }
225        }
226
227        let max_len = U256::from(u32::max_value() / 2);
228
229        if base_len > max_len || mod_len > max_len || exp_len > max_len {
230            return U256::max_value();
231        }
232        let (base_len, exp_len, mod_len) =
233            (base_len.low_u64(), exp_len.low_u64(), mod_len.low_u64());
234
235        // read fist 32-byte word of the exponent.
236        let exp_low = if base_len + 96 >= input.len() as u64 {
237            U256::zero()
238        } else {
239            let mut buf = [0; 32];
240            let mut reader =
241                input[(96 + base_len as usize)..].chain(io::repeat(0));
242            let len = min(exp_len, 32) as usize;
243            reader
244                .read_exact(&mut buf[(32 - len)..])
245                .expect("reading from zero-extended memory cannot fail; qed");
246            U256::from_big_endian(&buf[..])
247        };
248        // EIP-7883 doubles the per-byte cost of the exponent tail.
249        let exp_byte_factor = match self {
250            ModexpPricer::Osaka { .. } => 16,
251            _ => 8,
252        };
253        let iter_count =
254            max(Self::adjusted_exp_len(exp_len, exp_low, exp_byte_factor), 1);
255
256        match self {
257            ModexpPricer::Byzantium { divisor } => Self::byzantium_gas_calc(
258                base_len, mod_len, iter_count, *divisor,
259            ),
260            ModexpPricer::Berlin { base } => {
261                Self::berlin_gas_calc(base_len, mod_len, iter_count, *base)
262            }
263            ModexpPricer::Osaka { base } => {
264                Self::osaka_gas_calc(base_len, mod_len, iter_count, *base)
265            }
266        }
267    }
268}

270impl ModexpPricer {
271    pub fn byzantium_gas_calc(
272        base_len: u64, mod_len: u64, iter_count: u64, divisor: usize,
273    ) -> U256 {
274        let m = max(mod_len, base_len);
275        let (gas, overflow) =
276            Self::mult_complexity(m).overflowing_mul(iter_count);
277        if overflow {
278            return U256::max_value();
279        }
280        (gas / divisor as u64).into()
281    }
282
283    fn adjusted_exp_len(len: u64, exp_low: U256, byte_factor: u64) -> u64 {
284        let bit_index = if exp_low.is_zero() {
285            0
286        } else {
287            (255 - exp_low.leading_zeros()) as u64
288        };
289        if len <= 32 {
290            bit_index
291        } else {
292            byte_factor * (len - 32) + bit_index
293        }
294    }
295
296    fn mult_complexity(x: u64) -> u64 {
297        match x {
298            x if x <= 64 => x * x,
299            x if x <= 1024 => (x * x) / 4 + 96 * x - 3072,
300            x => (x * x) / 16 + 480 * x - 199680,
301        }
302    }
303
304    pub fn berlin_gas_calc(
305        base_len: u64, mod_len: u64, iter_count: u64, base_gas: usize,
306    ) -> U256 {
307        fn calculate_multiplication_complexity(
308            base_len: u64, mod_len: u64,
309        ) -> U256 {
310            let max_len = max(base_len, mod_len);
311            let mut words = max_len / 8;
312            if max_len % 8 > 0 {
313                words += 1;
314            }
315            let words = U256::from(words);
316            words * words
317        }
318
319        let multiplication_complexity =
320            calculate_multiplication_complexity(base_len, mod_len);
321        let gas = (multiplication_complexity * U256::from(iter_count))
322            / U256::from(3);
323        let gas_u64 = if gas >= U256::from(u64::MAX) {
324            u64::MAX
325        } else {
326            gas.as_u64()
327        };
328        max(base_gas as u64, gas_u64).into()
329    }
330
331    // CIP-174: EIP-7883. Inputs are bounded by EIP-7823 (checked in `cost`),
332    // so none of the arithmetic below can overflow u64.
333    pub fn osaka_gas_calc(
334        base_len: u64, mod_len: u64, iter_count: u64, base_gas: usize,
335    ) -> U256 {
336        let max_len = max(base_len, mod_len);
337        let mut words = max_len / 8;
338        if max_len % 8 > 0 {
339            words += 1;
340        }
341        let multiplication_complexity =
342            if max_len <= 32 { 16 } else { 2 * words * words };
343        let gas = multiplication_complexity * iter_count;
344        max(base_gas as u64, gas).into()
345    }
