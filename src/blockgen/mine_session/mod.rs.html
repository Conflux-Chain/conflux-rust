16// A maximum interval to update the mining block
17const BLOCK_FORCE_UPDATE_INTERVAL: Duration = Duration::from_secs(10);
18// Avoid busy loop: sleep for 30ms if the current loop iteration does nothing
19const BLOCKGEN_LOOP_SLEEP_DURATION: Duration = time::Duration::from_millis(30);
20
21pub(crate) struct MiningSession<'a> {
22    bg: &'a BlockGenerator,
23    miner: &'a dyn MineWorker,
24    solution_rx: SolutionReceiver,
25    state: MineState,
26}
27
28impl<'a> MiningSession<'a> {
29    pub fn new(
30        bg: &'a BlockGenerator, miner: &'a dyn MineWorker,
31        solution_rx: SolutionReceiver,
32    ) -> Self {
33        let state =
34            MineState::with_capacity(bg.pow_config.pow_problem_window_size);
35
36        MiningSession {
37            bg,
38            miner,
39            solution_rx,
40            state,
41        }
42    }
43
44    pub fn run(mut self) {
45        while self.bg.is_running() {
46            // TODO: #transations TBD
47            if !self.should_mine() {
48                thread::sleep(BLOCKGEN_LOOP_SLEEP_DURATION);
49                continue;
50            }
51
52            if self.is_mining_block_outdated() {
53                // Compute new mining task and send to workers
54                self.update_mining_task();
55            }
56            // Pull the mining solution
57            self.process_pending_solutions();
58        }
59    }
60
61    fn should_mine(&self) -> bool {
62        self.pow_config.test_mode || !self.sync.catch_up_mode()
63    }
64
65    /// Check if we need to mine on a new block
66    fn is_mining_block_outdated(&self) -> bool {
67        // TODO: update the logic here.
68
69        let Some(block) = self.state.mining_block() else {
70            return true;
71        };
72
73        // 1st Check: if the parent block changed
74        let best_block_hash = self.consensus.best_block_hash();
75        if best_block_hash != *block.block_header.parent_hash() {
76            return true;
77        }
78
79        // 2nd Check: if the last block is too old, we will generate a new
80        // block. Checking transaction updates and referees might be
81        // costly and the trade-off is unclear here. It is simple to
82        // just enforce a time here.
83
84        self.state.last_assemble_elapsed() > BLOCK_FORCE_UPDATE_INTERVAL
85    }
86
87    fn update_mining_task(&mut self) {
88        let next_mining_block = self.assembler.assemble_new_mining_block(None);
89        self.state.touch_last_assemble();
90
91        let problem = ProofOfWorkProblem::from_block_header(
92            &next_mining_block.block_header,
93        );
94
95        self.state.update_job(next_mining_block, problem);
96        self.send_problem(problem);
97    }
98
99    /// Pull mining solution from the workers, returns whether a valid solution
100    /// is found
101    fn process_pending_solutions(&mut self) {
102        while let Ok(solution) = self.solution_rx.try_recv() {
103            debug!("new solution: {:?}", solution);
104
105            if let Some(mut mined_block) =
106                self.state.validate_and_claim_solution(&solution, |p, s| {
107                    has_zero_high_nonce(s) && self.bg.pow.validate(p, s)
108                })
109            {
110                mined_block.block_header.set_nonce(solution.nonce);
111                mined_block.block_header.compute_hash();
112
113                self.on_mined_block(mined_block);
114                return;
115            } else {
116                warn!(
117                    "Received invalid solution from miner: nonce = {}!",
118                    &solution.nonce
119                );
120            }
121        }
122
123        // The solution channel is empty, or all pending solutions were invalid.
124        // We should ensure if the miner has current job.
125        self.ensure_miner_has_current_work();
126        // Avoid busy-waiting
127        thread::sleep(BLOCKGEN_LOOP_SLEEP_DURATION);
128    }
129
130    /// We will send out heartbeat because newcomers or
131    /// disconnected people may lose the previous message
132    fn ensure_miner_has_current_work(&mut self) {
133        if !self.pow_config.use_stratum() {
134            return;
135        }
136
137        if let Some(problem) = self.state.check_for_renotification() {
138            self.send_problem(problem);
139        }
140    }
141
142    fn send_problem(&mut self, problem: ProofOfWorkProblem) {
143        self.miner.receive_problem(problem);
144
145        trace!("send problem: {:?}", problem);
146        self.state.touch_last_notify();
147    }
148}
